file(GLOB_RECURSE lr_examples_common_sources CONFIGURE_DEPENDS common/src/*.cpp common/include/*.h)
add_library(lr_examples_common STATIC ${lr_examples_common_sources})
target_include_directories(lr_examples_common PUBLIC common/include)
target_link_libraries(lr_examples_common PUBLIC lr::render lr::asset lr::scene PRIVATE lr_build_options)

add_executable(lrender_cli cli/main.cpp)
target_link_libraries(lrender_cli PRIVATE lr_examples_common lr::present_image lr_build_options)

if(TARGET lr::present_sdl)
    add_executable(lrender_view view/main.cpp)
    target_link_libraries(lrender_view PRIVATE lr_examples_common lr::present_sdl lr_build_options)
    if(TARGET lr::present_vulkan)
        target_link_libraries(lrender_view PRIVATE lr::present_vulkan)
        target_compile_definitions(lrender_view PRIVATE LR_VIEW_VULKAN_PRESENTER)
    endif()
endif()
